/* Shared atoms used across all 3 variants */

const Logo = ({ color = "currentColor", size = 28, mode = "image" }) => {
  // mode: "image" uses the real PNG logo; "wordmark" uses the SVG/text version
  if (mode === "image") {
    const isLight = color !== "currentColor" && (color === "#FBF8F2" || color === "var(--paper)" || /^#[fF]/.test(color));
    return (
      <img
        src="assets/logo-cannolis.png"
        alt="Cannolis Experience"
        style={{
          height: size * 2.2,
          width: 'auto',
          display: 'block',
          filter: isLight ? 'invert(1) brightness(1.05)' : 'none',
        }}
      />
    );
  }
  return (
    <div style={{ display: 'flex', alignItems: 'center', gap: 10, color }}>
      <svg width={size} height={size} viewBox="0 0 32 32" fill="none">
        <ellipse cx="16" cy="16" rx="13" ry="6" stroke={color} strokeWidth="1.2" transform="rotate(-20 16 16)"/>
        <ellipse cx="9" cy="13.5" rx="2.5" ry="1.4" fill={color} transform="rotate(-20 9 13.5)"/>
        <ellipse cx="23" cy="18.5" rx="2.5" ry="1.4" fill={color} transform="rotate(-20 23 18.5)"/>
      </svg>
      <div style={{ display: 'flex', flexDirection: 'column', lineHeight: 1 }}>
        <span style={{ fontFamily: 'var(--script)', fontSize: 26, lineHeight: 0.8 }}>Cannolis</span>
        <span style={{ fontFamily: 'var(--sans)', fontSize: 9, letterSpacing: '0.32em', textTransform: 'uppercase', marginTop: 2 }}>Experience</span>
      </div>
    </div>
  );
};

const Eyebrow = ({ children, accent = false, num }) => (
  <div style={{ display: 'flex', alignItems: 'center', gap: 12 }}>
    {num && <span style={{ fontFamily: 'var(--mono)', fontSize: 10, color: 'var(--accent-deep)' }}>{num}</span>}
    <div style={{ width: 24, height: 1, background: accent ? 'var(--accent)' : 'var(--ink-muted)' }} />
    <span className="eyebrow" style={accent ? { color: 'var(--accent-deep)' } : {}}>{children}</span>
  </div>
);

const Mono = ({ children, ...rest }) => (
  <span className="mono" {...rest}>{children}</span>
);

const PhotoFrame = ({ src, alt, caption, ratio = "4/5", monoCaption, style }) => (
  <figure style={{ margin: 0, ...style }}>
    <div className="img-card" style={{ aspectRatio: ratio, background: 'var(--bone)' }}>
      <img src={src} alt={alt} />
    </div>
    {(caption || monoCaption) && (
      <figcaption style={{ display: 'flex', justifyContent: 'space-between', marginTop: 10, fontFamily: 'var(--mono)', fontSize: 10, color: 'var(--ink-muted)', letterSpacing: '0.06em' }}>
        <span>{monoCaption}</span>
        <span>{caption}</span>
      </figcaption>
    )}
  </figure>
);
